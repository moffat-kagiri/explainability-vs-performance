# Test set evaluation
poisson_deviance <- function(y, mu) { 
  2*mean(ifelse(y>0, y*log(y/mu), 0) - (y-mu))
}

test_matrix <- prep_matrix(test)
mu_test <- predict(final_model, test_matrix)*test$Exposure

cat(sprintf("\nTest Poisson deviance: %.5f\n", poisson_deviance(test$ClaimNb, mu_test)))

            sum(mu_test), sum(test$ClaimNb), sum(mu_test)/sum(test$ClaimNb)))
